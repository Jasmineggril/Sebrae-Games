# 🚀 CÓDIGO COMPLETO (SUBSTITUI TUDO)

```html id="gamepro01"
<!DOCTYPE html>
<html lang="pt-br">
<head>
<meta charset="UTF-8">
<title>Fazenda Brasil</title>

<link href="https://fonts.googleapis.com/css2?family=Poppins:wght@300;500;700&display=swap" rel="stylesheet">

<style>
body {
  margin: 0;
  font-family: 'Poppins', sans-serif;
  background: linear-gradient(135deg, #2e7d32, #66bb6a);
  color: white;
  text-align: center;
}

/* telas */
.screen {
  display: none;
  padding: 20px;
}

.active {
  display: block;
}

/* loading */
#loading {
  display: flex;
  flex-direction: column;
  justify-content: center;
  height: 100vh;
}

.loader {
  width: 200px;
  height: 10px;
  background: #ffffff33;
  margin: 20px auto;
  border-radius: 10px;
  overflow: hidden;
}

.loader-fill {
  height: 100%;
  width: 0%;
  background: #a5d6a7;
  transition: width 0.3s;
}

/* menu */
#menu h1 {
  font-size: 32px;
}

.logo {
  font-size: 40px;
}

/* jogo */
.container {
  max-width: 400px;
  margin: auto;
}

.card {
  background: rgba(255,255,255,0.15);
  backdrop-filter: blur(10px);
  padding: 15px;
  border-radius: 15px;
  margin: 15px 0;
}

button {
  width: 100%;
  padding: 12px;
  margin: 5px 0;
  border: none;
  border-radius: 10px;
  font-weight: bold;
  cursor: pointer;
}

.btn-plant { background: #43a047; }
.btn-organic { background: #81c784; }
.btn-chemical { background: #ef5350; }
.btn-harvest { background: #ffca28; color: black; }

</style>
</head>

<body>

<!-- LOADING -->
<div id="loading" class="screen active">
  <h1>🌾 Fazenda Brasil</h1>
  <p>Carregando...</p>
  <div class="loader">
    <div class="loader-fill" id="loaderFill"></div>
  </div>
</div>

<!-- MENU -->
<div id="menu" class="screen">
  <h1 class="logo">🌾 Fazenda Brasil</h1>
  <p>Safra Inteligente</p>

  <button onclick="startGame()">▶️ Jogar</button>

  <p style="margin-top:20px; font-size:14px;">
    Desenvolvido por você 🚀
  </p>
</div>

<!-- GAME -->
<div id="game" class="screen">
  <div class="container">

    <h2>🌱 Sua Fazenda</h2>

    <div class="card">
      <p>💰 Moedas: <span id="money">0</span></p>
      <p>🌱 Solo: <span id="soil">100</span></p>
      <p id="status">Plante algo!</p>
    </div>

    <div class="card">
      <button class="btn-plant" onclick="plant()">🌱 Plantar</button>
      <button class="btn-organic" onclick="organic()">🌿 Orgânico</button>
      <button class="btn-chemical" onclick="chemical()">⚡ Químico</button>
      <button class="btn-harvest" onclick="harvest()">🌾 Colher</button>
    </div>

  </div>
</div>

<script>

/* ===== LOADING ===== */
let progress = 0;
let loader = document.getElementById("loaderFill");

let loadingInterval = setInterval(() => {
  progress += 10;
  loader.style.width = progress + "%";

  if (progress >= 100) {
    clearInterval(loadingInterval);
    showScreen("menu");
  }
}, 200);

/* ===== TROCA DE TELAS ===== */
function showScreen(id) {
  document.querySelectorAll(".screen").forEach(s => s.classList.remove("active"));
  document.getElementById(id).classList.add("active");
}

function startGame() {
  showScreen("game");
}

/* ===== GAME ===== */
let money = 0;
let soil = 100;
let planted = false;
let strategy = "";

function updateUI() {
  document.getElementById("money").innerText = money;
  document.getElementById("soil").innerText = soil;
}

function plant() {
  planted = true;
  document.getElementById("status").innerText = "🌱 Plantado!";
}

function organic() {
  if (!planted) return;
  strategy = "organic";
  document.getElementById("status").innerText = "🌿 Orgânico";
}

function chemical() {
  if (!planted) return;
  strategy = "chemical";
  soil -= 10;
  document.getElementById("status").innerText = "⚡ Químico";
  updateUI();
}

function harvest() {
  if (!planted) return;

  let gain = 10;

  if (strategy === "chemical") gain += 5;
  if (strategy === "organic") soil += 5;

  if (soil < 50) gain -= 3;

  money += gain;

  let events = [
    "🌧️ Chuva ajudou!",
    "☀️ Seca prejudicou!",
    "🐛 Praga apareceu!"
  ];

  alert(events[Math.floor(Math.random()*events.length)]);

  alert("📊 O Brasil é líder agrícola mundial!");

  planted = false;
  strategy = "";

  updateUI();
  document.getElementById("status").innerText = "🌾 Colhido!";
}

updateUI();

</script>

</body>
</html>
```
