# 🚀 CÓDIGO PROFISSIONAL (COLA NO SEU index.html)

👉 substitui tudo pelo código abaixo:

```html
<!DOCTYPE html>
<html lang="pt-br">
<head>
<meta charset="UTF-8">
<title>Fazenda Brasil: Safra Inteligente</title>

<link href="https://fonts.googleapis.com/css2?family=Poppins:wght@300;500;700&display=swap" rel="stylesheet">

<style>
body {
  font-family: 'Poppins', sans-serif;
  background: linear-gradient(135deg, #2e7d32, #66bb6a);
  color: #fff;
  text-align: center;
  margin: 0;
  padding: 0;
}

.container {
  max-width: 400px;
  margin: auto;
  padding: 20px;
}

h1 {
  margin-top: 20px;
}

.card {
  background: rgba(255,255,255,0.15);
  backdrop-filter: blur(10px);
  padding: 15px;
  border-radius: 15px;
  margin: 15px 0;
}

.status {
  font-weight: bold;
  margin-top: 10px;
}

button {
  width: 100%;
  padding: 12px;
  margin: 5px 0;
  border: none;
  border-radius: 10px;
  font-size: 16px;
  font-weight: bold;
  cursor: pointer;
  transition: 0.2s;
}

button:hover {
  transform: scale(1.05);
}

.btn-plant { background: #43a047; color: white; }
.btn-organic { background: #81c784; }
.btn-chemical { background: #ef5350; }
.btn-harvest { background: #ffca28; color: black; }

.tutorial {
  font-size: 14px;
  text-align: left;
}

</style>
</head>

<body>

<div class="container">

<h1>🌾 Fazenda Brasil</h1>
<p>Safra Inteligente</p>

<div class="card">
  <p>💰 Moedas: <span id="money">0</span></p>
  <p>🌱 Solo: <span id="soil">100</span></p>
  <p class="status" id="status">Plante algo!</p>
</div>

<div class="card">
  <button class="btn-plant" onclick="plant()">🌱 Plantar</button>
  <button class="btn-organic" onclick="organic()">🌿 Orgânico</button>
  <button class="btn-chemical" onclick="chemical()">⚡ Químico</button>
  <button class="btn-harvest" onclick="harvest()">🌾 Colher</button>
</div>

<div class="card tutorial">
  <b>📘 Tutorial:</b><br><br>
  🌱 Plante uma cultura<br>
  🌿 Escolha orgânico (sustentável)<br>
  ⚡ ou químico (mais rápido)<br>
  🌾 Colha e ganhe dinheiro<br>
  💡 Cuide do solo para produzir melhor!
</div>

</div>

<script>

let money = 0;
let soil = 100;
let planted = false;
let strategy = "";

function updateUI() {
  document.getElementById("money").innerText = money;
  document.getElementById("soil").innerText = soil;
}

function setStatus(msg) {
  document.getElementById("status").innerText = msg;
}

function plant() {
  planted = true;
  setStatus("🌱 Plantado!");
}

function organic() {
  if (!planted) return;
  strategy = "organic";
  setStatus("🌿 Cultivo orgânico selecionado");
}

function chemical() {
  if (!planted) return;
  strategy = "chemical";
  soil -= 10;
  setStatus("⚡ Cultivo químico selecionado");
  updateUI();
}

function harvest() {
  if (!planted) return;

  let gain = 10;

  if (strategy === "chemical") gain += 5;
  if (strategy === "organic") soil += 5;

  if (soil < 50) gain -= 3;

  money += gain;

  let events = [
    "🌧️ Chuva ajudou sua plantação!",
    "☀️ Seca reduziu sua produção!",
    "🐛 Praga atacou parte da safra!"
  ];

  let event = events[Math.floor(Math.random() * events.length)];

  alert(event);

  alert("📊 Ficha do Produtor:\nO Brasil é um dos maiores produtores agrícolas do mundo!");

  planted = false;
  strategy = "";

  updateUI();
  setStatus("🌾 Colheita realizada!");
}

updateUI();

</script>

</body>
</html>
```
